export const debounce = (func, delay = 300) => {
  let timer;

  return (...args) => {
    clearTimeout(timer);
    timer = setTimeout(() => {
      func(...args);
    }, delay);
  };
};


import { useEffect, useMemo, useState } from "react";
import { debounce } from "../utils/debounce";
import axiosInstance from "../api/axiosInstance";

const ChatSearch = ({ chats }) => {
  const [search, setSearch] = useState("");
  const [filteredChats, setFilteredChats] = useState(chats);
  const [apiResults, setApiResults] = useState([]);

  // 🔹 1. Frontend filtering (instant)
  useEffect(() => {
    const localFiltered = chats.filter((chat) =>
      chat.name.toLowerCase().includes(search.toLowerCase())
    );
    setFilteredChats(localFiltered);
  }, [search, chats]);

  // 🔹 2. API search (debounced)
  const debouncedApiCall = useMemo(
    () =>
      debounce(async (query) => {
        if (!query) return;

        try {
          const res = await axiosInstance.get(`/chats?search=${query}`);
          setApiResults(res.data);
        } catch (err) {
          console.log("API search error", err);
        }
      }, 400),
    []
  );

  useEffect(() => {
    debouncedApiCall(search);
  }, [search, debouncedApiCall]);

  return (
    <div>
      <input
        className="form-control my-2"
        placeholder="Search chats"
        value={search}
        onChange={(e) => setSearch(e.target.value)}
      />

      <h6>Local Results</h6>
      {filteredChats.map((chat) => (
        <div key={chat._id}>{chat.name}</div>
      ))}

      <h6>API Results</h6>
      {apiResults.map((chat) => (
        <div key={chat._id}>{chat.name}</div>
      ))}
    </div>
  );
};

export default ChatSearch;
